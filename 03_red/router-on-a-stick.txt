! =====================================================================
!  R-HQ - Cisco 2911 (Packet Tracer) - hace el papel de pfSense
!  Gi0/0      -> trunk a SW-CORE (subinterfaces por VLAN)
!  Gi0/1      -> enlace WAN hacia R-ISP  203.0.113.10/24
!  Las ACL reproducen las reglas de pfSense (minimo privilegio).
! =====================================================================
enable
configure terminal
hostname R-HQ
no ip domain-lookup
enable secret Lab#Router2026
service password-encryption

! ---- Subinterfaces 802.1Q ----
interface GigabitEthernet0/0
 no ip address
 no shutdown
exit
interface GigabitEthernet0/0.10
 description VLAN10-DMZ
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.0
 ip nat inside
exit
interface GigabitEthernet0/0.20
 description VLAN20-INTERNA
 encapsulation dot1Q 20
 ip address 10.10.20.1 255.255.255.0
 ip nat inside
exit
interface GigabitEthernet0/0.30
 description VLAN30-IOT
 encapsulation dot1Q 30
 ip address 10.10.30.1 255.255.255.0
 ip nat inside
exit
interface GigabitEthernet0/0.99
 description VLAN99-GESTION
 encapsulation dot1Q 99
 ip address 10.10.199.1 255.255.255.0
exit
interface GigabitEthernet0/0.999
 encapsulation dot1Q 999 native
exit

interface GigabitEthernet0/1
 description WAN-a-ISP
 ip address 203.0.113.10 255.255.255.0
 ip nat outside
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 203.0.113.1

! ---- DHCP (excluye gateways y servidores estaticos) ----
ip dhcp excluded-address 10.10.20.1 10.10.20.99
ip dhcp excluded-address 10.10.30.1 10.10.30.9
ip dhcp pool INTERNA
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 10.10.20.1
exit
ip dhcp pool IOT
 network 10.10.30.0 255.255.255.0
 default-router 10.10.30.1
 dns-server 10.10.30.1
exit

! =====================================================================
!  ACL de segmentacion (equivalentes a las reglas pfSense R1-R12)
! =====================================================================

! --- Entrante desde Internet (Gi0/1 in) ---
ip access-list extended WAN-IN
 remark R2 IPsec desde la sucursal / cliente remoto
 permit udp any host 203.0.113.10 eq isakmp
 permit udp any host 203.0.113.10 eq 4500
 permit esp any host 203.0.113.10
 remark R1 servicios publicados en DMZ: la ACL entrante se evalua ANTES del NAT,
 remark por eso el destino es la IP publica (NAT estatico -> 10.10.10.20 / .30)
 permit tcp any host 203.0.113.10 eq www
 permit tcp any host 203.0.113.10 eq 3000
 remark trafico ya descifrado del tunel IPsec (IOS lo vuelve a evaluar)
 permit ip 192.168.50.0 0.0.0.255 10.10.20.0 0.0.0.255
 remark trafico de retorno de sesiones iniciadas desde dentro
 permit tcp any any established
 permit icmp any any echo-reply
 remark R3 todo lo demas, incluida la red interna, se bloquea
 deny ip any any log
exit

! --- Desde la DMZ (Gi0/0.10 in) ---
ip access-list extended DMZ-IN
 remark R4 solo la API consulta la BD interna
 permit tcp host 10.10.10.30 host 10.10.20.50 eq 3306
 remark R5 bloquear pivoteo hacia interna e IoT
 deny ip 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255 log
 deny ip 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255 log
 permit udp 10.10.10.0 0.0.0.255 host 10.10.10.1 eq domain
 permit tcp 10.10.10.0 0.0.0.255 any established
 permit tcp 10.10.10.0 0.0.0.255 any eq 443
 deny ip any any log
exit

! --- Desde IoT (Gi0/0.30 in) ---
ip access-list extended IOT-IN
 permit udp 10.10.30.0 0.0.0.255 host 10.10.30.1 eq domain
 permit udp 10.10.30.0 0.0.0.255 any eq bootps
 remark R7 aislar IoT de cualquier red privada
 deny ip 10.10.30.0 0.0.0.255 10.0.0.0 0.255.255.255 log
 deny ip 10.10.30.0 0.0.0.255 172.16.0.0 0.15.255.255 log
 deny ip 10.10.30.0 0.0.0.255 192.168.0.0 0.0.255.255 log
 remark R6 salida solo HTTPS y MQTT-TLS a la nube del fabricante
 permit tcp 10.10.30.0 0.0.0.255 any eq 443
 permit tcp 10.10.30.0 0.0.0.255 any eq 8883
 deny ip any any log
exit

! --- Desde la red interna (Gi0/0.20 in) ---
ip access-list extended INTERNA-IN
 remark Trafico VPN de retorno hacia la sucursal (debe ir antes de los deny)
 permit ip 10.10.20.0 0.0.0.255 192.168.50.0 0.0.0.255
 remark R8/R9 DNS solo hacia el gateway (filtro antiphishing)
 permit udp 10.10.20.0 0.0.0.255 host 10.10.20.1 eq domain
 deny udp 10.10.20.0 0.0.0.255 any eq domain log
 deny tcp 10.10.20.0 0.0.0.255 any eq domain log
 remark no iniciar conexiones hacia IoT
 deny ip 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255 log
 permit udp 10.10.20.0 0.0.0.255 any eq bootps
 remark R10 navegacion web y acceso a la DMZ
 permit tcp 10.10.20.0 0.0.0.255 any eq www
 permit tcp 10.10.20.0 0.0.0.255 any eq 443
 permit tcp 10.10.20.0 0.0.0.255 host 10.10.10.30 eq 3000
 permit tcp host 10.10.20.50 host 10.10.10.30 established
 permit icmp 10.10.20.0 0.0.0.255 any echo
 deny ip any any log
exit

interface GigabitEthernet0/1
 ip access-group WAN-IN in
exit
interface GigabitEthernet0/0.10
 ip access-group DMZ-IN in
exit
interface GigabitEthernet0/0.20
 ip access-group INTERNA-IN in
exit
interface GigabitEthernet0/0.30
 ip access-group IOT-IN in
exit

! ---- NAT: estatico para la DMZ, PAT para el resto (excluyendo trafico VPN) ----
ip nat inside source static tcp 10.10.10.20 80 203.0.113.10 80
ip nat inside source static tcp 10.10.10.30 3000 203.0.113.10 3000
ip access-list extended NAT-PAT
 deny ip 10.10.20.0 0.0.0.255 192.168.50.0 0.0.0.255
 permit ip 10.10.0.0 0.0.255.255 any
exit
ip nat inside source list NAT-PAT interface GigabitEthernet0/1 overload

! ---- Administracion segura ----
ip domain-name lab.local
username admin privilege 15 secret Lab#Admin2026
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
access-list 10 permit 10.10.199.0 0.0.0.255
line vty 0 4
 access-class 10 in
 transport input ssh
 login local
exit

end
write memory
