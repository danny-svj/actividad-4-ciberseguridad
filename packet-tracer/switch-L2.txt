! =====================================================================
!  SW-CORE - Cisco 2960 (Packet Tracer)
!  Gi0/1  -> trunk hacia R-HQ (router-on-a-stick, Gi0/0)
!  Fa0/1-5   VLAN 10 DMZ      (DVWA 10.10.10.20, API 10.10.10.30)
!  Fa0/6-15  VLAN 20 INTERNA  (PCs por DHCP, servidor BD 10.10.20.50)
!  Fa0/16-20 VLAN 30 IOT      (camaras / sensores por DHCP)
!  Fa0/21-24 sin uso -> VLAN 999 (black hole) y apagados
! =====================================================================
enable
configure terminal
hostname SW-CORE
no ip domain-lookup
enable secret Lab#Switch2026
service password-encryption
banner motd # Acceso solo para personal autorizado. Actividad monitoreada. #

vlan 10
 name DMZ
vlan 20
 name INTERNA
vlan 30
 name IOT
vlan 99
 name GESTION
vlan 999
 name BLACKHOLE
exit

! ---- Trunk hacia el router: solo las VLAN necesarias, VLAN nativa no usada ----
interface GigabitEthernet0/1
 description TRUNK-a-R-HQ
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99
 switchport nonegotiate
 no shutdown
exit

! ---- Puertos de acceso ----
interface range FastEthernet0/1 - 5
 description DMZ
 switchport mode access
 switchport access vlan 10
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

interface range FastEthernet0/6 - 15
 description INTERNA
 switchport mode access
 switchport access vlan 20
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

interface range FastEthernet0/16 - 20
 description IOT
 switchport mode access
 switchport access vlan 30
 switchport nonegotiate
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
exit

interface range FastEthernet0/21 - 24
 description SIN-USO
 switchport mode access
 switchport access vlan 999
 shutdown
exit

! ---- Gestion del switch en VLAN 99 ----
interface Vlan99
 ip address 10.10.199.2 255.255.255.0
 no shutdown
exit
! NOTA: 10.10.199.0/24 = gestion; no se solapa con el pool VPN 10.10.99.0/24
ip default-gateway 10.10.199.1

! ---- SSH para administracion ----
ip domain-name lab.local
username admin privilege 15 secret Lab#Admin2026
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line vty 0 15
 transport input ssh
 login local
exit
line console 0
 login local
exit

end
write memory

! Verificacion:
!   show vlan brief
!   show interfaces trunk
!   show port-security
